! Architecture note: NAT is NOT performed on this switch (ipbasek9 does not
! support it). An Orange Pi running Ubuntu acts as the NAT router:
!   - Orange Pi WAN (eth0) -> phone hotspot / venue internet drop
!   - Orange Pi LAN (eth1/USB dongle) -> Gi1/0/7 (VLAN100 access port)
! The switch handles all inter-VLAN routing and ACLs; internet-bound traffic
! is forwarded to the Orange Pi via the default route (10.0.100.10).

! ===== Former WAN uplink VLAN =====
! Port 48 is no longer the WAN uplink. WAN is now handled by the Orange Pi
! (see README: Orange Pi NAT Router Setup). VLAN200 is left unaddressed;
! port 48 can be used as a spare access port or left disconnected.
interface Vlan200
 description Unused-FormerWAN
 no ip address
!
! ===== Internet-only VLAN =====
! Ports 44-47. Devices here get internet access but are completely
! blocked from reaching any FMS or team network addresses.
interface Vlan300
 description Internet-Only
 ip address 10.0.200.4 255.255.255.0
 ip access-group INTERNET-ONLY in
!
ip classless
ip http server
ip http secure-server
!
! ===== Team / FMS access list =====
! Applied inbound on team VLAN interfaces (Vlan10-60).
! Allows FMS comms and DHCP. Internet access is provided via the
! Orange Pi NAT router (10.0.100.10) through the switch default route.
ip access-list extended DS-FMS
 permit udp any eq 1145 10.0.100.0 0.0.0.255 eq 1160
 permit tcp any 10.0.100.0 0.0.0.255 eq 1750
 permit icmp any 10.0.100.0 0.0.0.255
 permit icmp any 10.0.0.4 0.255.255.0
 permit udp any any eq bootpc
 permit udp any any eq bootps
 permit ip any any
!
! ===== Internet-only access list =====
! Applied inbound on Vlan300 (ports 44-47).
! DHCP is allowed first so devices can get an address.
! Then all RFC-1918 private ranges are blocked (no access to FMS, team
! networks, or anything else internal). Everything else (the internet) is
! allowed and forwarded to the Orange Pi NAT router via the default route.
ip access-list extended INTERNET-ONLY
 permit udp any any eq bootpc
 permit udp any any eq bootps
 deny ip any 10.0.0.0 0.255.255.255
 deny ip any 172.16.0.0 0.15.255.255
 deny ip any 192.168.0.0 0.0.255.255
 permit ip any any
